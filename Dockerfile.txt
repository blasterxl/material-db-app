FROM node:18-slim

# Встановлюємо Python та системні залежності
RUN apt-get update && apt-get install -y \
    python3 \
    python3-pip \
    python3-venv \
    && rm -rf /var/lib/apt-get/lists/*

WORKDIR /app

# Копіюємо package.json та встановлюємо Node.js пакети
COPY package*.json ./
RUN npm install --production

# Встановлюємо Python-бібліотеки
RUN pip3 install reportlab Pillow --break-system-packages

# Копіюємо весь вміст папки проекту
COPY . .

EXPOSE 4177

CMD ["node", "server.js"]
